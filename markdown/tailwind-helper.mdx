## TailwindHelper
A utility component that displays the current breakpoint. Only visible in development.

### Installation
<InstallCommand url="https://registry.rchoudhury.dev/r/tailwind-helper.json" />

### Usage
```tsx
import { TailwindHelper } from "@/components/tw-helper";

export default function Layout({ children }) {
  return (
    <>
      {children}
      <TailwindHelper />
    </>
  );
}
```
