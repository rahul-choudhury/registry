## Multi Step Form
Component and hook for managing multi step forms.

### Installation
<InstallCommand url="https://registry.rchoudhury.dev/r/multi-step-form.json" />

### Usage
```tsx
// page.tsx
import { MultiStep } from "@/components/multi-step-form";

export default function Page() {
  return (
    <MultiStep>
      <Step1 />
      <Step2 />
      ...
    </MultiStep>
  );
}

// step1.tsx
import { useMultiStep } from "@/components/multi-step-form";

export function Step1() {
  const { nextStep, prevStep, isFirstStep, isLastStep } = useMultiStep();

  return (
    <form>
      ...
      <button type="button" disabled={isFirstStep} onClick={prevStep}>
        Previous
      </button>
      <button type="button" disabled={isLastStep} onClick={nextStep}>
        Next
      </button>
    </form>
  );
}
```
